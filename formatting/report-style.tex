% Arial 12, with no outer indentation; meaningful spaces inside code remain.
% Long lines wrap visually without changing the executable source code.

  fontsize=\normalsize,fontfamily=\rmdefault,
  formatcom=\addfontfeatures{Ligatures=TeXOff},
  baselinestretch=1.5,breaklines,breakanywhere,
  breakautoindent=false,breakindent=0pt,
  breaksymbolleft={},breaksymbolright={},
  xleftmargin=0pt,xrightmargin=0pt,commandchars=\\\{\}}

% The edited reference has a left-aligned title followed immediately by the TOC.
% The repository is appended to the date by report-layout.lua, before the TOC.
\makeatletter
\renewcommand{\maketitle}{%
  \begingroup
  \parindent=0pt
  {\raggedright\setstretch{1}\fontsize{28}{34}\selectfont\@title\par}%
  \vspace{4pt}%
  {\raggedright\setstretch{1}\fontsize{12}{18}\selectfont\@author\par}%
  {\raggedright\setstretch{1}\fontsize{12}{18}\selectfont\@date\par}%
  \endgroup
  \vspace{6pt}%
}
% Dotted leaders and three TOC levels, as in the supplied Word reference.
\renewcommand*{\l@section}[2]{\begingroup\bfseries\@dottedtocline{1}{0pt}{1.5em}{{\bfseries #1}}{{\bfseries #2}}\endgroup}
\renewcommand*{\l@subsection}[2]{\begingroup\bfseries\@dottedtocline{2}{0.1667in}{2.3em}{{\bfseries #1}}{{\bfseries #2}}\endgroup}
\renewcommand*{\l@subsubsection}[2]{\begingroup\bfseries\@dottedtocline{3}{0.3333in}{3.2em}{{\bfseries #1}}{{\bfseries #2}}\endgroup}
\makeatother
